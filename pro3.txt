<button id="stringBtn" onclick="manipulateString()">Manipulate String</button>

<h2>Array Manipulation</h2>
<label>Enter Array Items separated by commas:</label>
<input type="text" id="arrayInput" placeholder="Example: Apple, Banana, Mango, Orange">
<button id="arrayBtn" onclick="manipulateArray()">Manipulate Array</button>
<p id="arrayOutput">Array results will appear here.</p>

<script>
function manipulateString(){
let t=stringInput.value.trim();
if(!t){
stringOutput.innerHTML="Please enter a string.";
return;
}
stringOutput.innerHTML=
"<b>Original String:</b> "+t+"<br>"+
"<b>Uppercase:</b> "+t.toUpperCase()+"<br>"+
"<b>Lowercase:</b> "+t.toLowerCase()+"<br>"+
"<b>Length:</b> "+t.length+"<br>"+
"<b>Reversed String:</b> "+t.split("").reverse().join("")+"<br>"+
"<b>Number of Words:</b> "+t.split(/\s+/).length;
}
function manipulateArray(){
let i=arrayInput.value.trim();
if(!i){
arrayOutput.innerHTML="Please enter array items.";
return;
}
let a=i.split(",").map(x=>x.trim()).filter(x=>x),
s=[...a].sort(),
r=[...a].reverse();
a.push("New Item");
let p=a.pop();
arrayOutput.innerHTML=
"<b>Original Array:</b> "+i+"<br>"+
"<b>Array Elements:</b> "+a.join(", ")+"<br>"+
"<b>Total Elements:</b> "+a.length+"<br>"+
"<b>Sorted Array:</b> "+s.join(", ")+"<br>"+
"<b>Reversed Array:</b> "+r.join(", ")+"<br>"+
"<b>First Element:</b> "+a[0]+"<br>"+
"<b>Last Element:</b> "+a[a.length-1]+"<br>"+
"<b>Push Operation:</b> New Item added temporarily<br>"+
"<b>Pop Operation:</b> Removed Item is "+p;
}
</script>
