<div class="box">
<h2>Guess the Number</h2>
<p>I picked a number between <b>1 and 100</b>.</p>
<input type="number" id="guessInput" placeholder="Enter your guess">
<br>
<button id="guess" onclick="checkGuess()">Guess</button>
<button id="new" onclick="resetGame()">New Game</button>
<p id="message">Make your first guess!</p>
<p id="attempts">Attempts: 0</p>
</div>
<script>
let random=Math.floor(Math.random()*100)+1,
    count=0;
function checkGuess(){
let g=guessInput.value;
if(g==""||g<1||g>100){
message.innerHTML="Please enter a number from 1 to 100.";
return;
}
count++;
attempts.innerHTML="Attempts: "+count;
if(g==random)
message.innerHTML="Correct! You guessed it in "+count+" attempts.";
else if(g<random)
message.innerHTML="Too Low! Try a higher number.";
else
message.innerHTML="Too High! Try a lower number.";
guessInput.value="";
guessInput.focus();
}
function resetGame(){
random=Math.floor(Math.random()*100)+1;
count=0;
guessInput.value="";
message.innerHTML="New game started! Make your first guess.";
attempts.innerHTML="Attempts: 0";
guessInput.focus();
}
</script>
