<script>
let imgs=document.querySelectorAll(".slider img"),i=0;

function show(n){
    imgs[i].classList.remove("active");
    i=(n+imgs.length)%imgs.length;
    imgs[i].classList.add("active");
}

function next(){
    show(i+1);
}

function prev(){
    show(i-1);
}

setInterval(next,3000);
