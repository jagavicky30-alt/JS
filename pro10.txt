<h2>Fetch API Demo</h2>
<button onclick="fetchData()">Fetch Posts</button>
<button onclick="clearData()">Clear</button>

<p id="status">Click Fetch Posts</p>
<div id="grid"></div>

<script>
async function fetchData(){
    let grid=document.getElementById("grid");
    let status=document.getElementById("status");
    status.innerHTML="Loading...";
    grid.innerHTML="";

    try{
        let res=await fetch("https://jsonplaceholder.typicode.com/posts");
        let data=await res.json();

        status.innerHTML=data.length+" posts fetched.";

        data.slice(0,10).forEach(p=>{
            grid.innerHTML+=`
            <div class="card">
            <h3>${p.title}</h3>
            <p>${p.body}</p>
            <small>Post ${p.id} | User ${p.userId}</small>
            </div>`;
        });
    }
    catch(e){
        status.innerHTML="Error loading data";
    }
}

function clearData(){
    document.getElementById("grid").innerHTML="";
    document.getElementById("status").innerHTML="Click Fetch Posts";
}
</script>
